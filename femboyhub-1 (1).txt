  local function new(cls, props, parent) local o = Instance.new(cls)
  for k, v in pairs(props) do if k ~= “Parent” then o[k] = v end end if
  parent then o.Parent = parent end return o end local function
  corner(r, p) return new(“UICorner”, { CornerRadius = UDim.new(0, r)
  }, p) end local function stroke(c, t, p) return new(“UIStroke”, {

  Enum.ApplyStrokeMode.Border }, p) end local function pad(a, b, c, d,
  p) return new(“UIPadding”, { PaddingLeft = UDim.new(0,a),

local function serialize() local out = {} for k, v in pairs(Flags) do if
typeof(v) == “Color3” then out[k] = { __c = true, v:ToHex() } elseif
typeof(v) == “EnumItem” then out[k] = { __e = true, v.Name,
tostring(v.EnumType) } else out[k] = v end end return out end

local function save_config(name) if not name or name == “” then return
false end pcall(writefile, CFGDIR .. “/” .. name .. “.json”,
HttpService:JSONEncode(serialize())) return true end

local function load_config(name) local ok, raw = pcall(readfile, CFGDIR
.. “/” .. name .. “.json”) if not ok then return false end local ok2,
data = pcall(HttpService.JSONDecode, HttpService, raw) if not ok2 then
return false end for k, v in pairs(data) do if type(v) == “table” and
v.__c then Flags[k] = Color3.fromHex(v.v) elseif type(v) == “table” and
v.__e then local e = Enum[v[2]]; if e then Flags[k] = e[v[1]] end else
Flags[k] = v end end return true end

local function list_configs() local ok, items = pcall(listfiles, CFGDIR)
local out = {} if not ok then return out end for _, f in ipairs(items)
do local n = f:match(“([^/\\]+)%.json$”) if n then out[#out + 1] = n end
end return out end

local OCTANT_DIRS = { Vector3.new(1,0,0), Vector3.new(-1,0,0),
Vector3.new(0,1,0), Vector3.new(0,-1,0), Vector3.new(0,0,1),

local function raycastIgnore(origin, target, list) local p =
RaycastParams.new() p.FilterType = Enum.RaycastFilterType.Exclude
p.FilterDescendantsInstances = list p.IgnoreWater = true return
Workspace:Raycast(origin, target - origin, p) end

local function wallThickness(origin, target) local p =
RaycastParams.new() p.FilterType = Enum.RaycastFilterType.Exclude
p.FilterDescendantsInstances = { LP.Character } p.IgnoreWater = true
local hit = Workspace:Raycast(origin, (target - origin).Unit * 600, p)
if not hit then return 0 end if target.Parent and
hit.Instance:IsDescendantOf(target.Parent) then return 0 end local deep
= Workspace:Raycast(hit.Position + hit.Normal * 0.1, hit.Normal * 12, p)
if deep then return (deep.Position - hit.Position).Magnitude end return
12 end

local function wallbangOffset(origin, target) local best, bestScore =
target, 0 for _, dir in ipairs(OCTANT_DIRS) do for step = 1, 3 do local
probe = target + dir * (step * 0.55) local th = wallThickness(origin,
probe) if th <= Flags.wall_thickness then local score = 1 / math.max(th,
0.05) if score > bestScore then bestScore = score; best = probe end end
end end return best end

local function manipulate(pos, origin) if not Flags.manip then return
pos end local best, bestTh = pos, math.huge local n = math.max(4,
math.floor(Flags.manip_samples)) for i = 1, n do local a = (i / n) *
math.pi * 2 local probe = pos + Vector3.new(math.cos(a), math.sin(a) *
0.6, math.sin(a)) * Flags.manip_radius * 0.15 local th =
wallThickness(origin, probe) if th < bestTh then bestTh = th; best =
probe end end return best end

local function backtrackPos(plr, nowPos) if not Flags.backtrack then
return nowPos end local buf = History[plr] if not buf or #buf < 2 then
return nowPos end local want = tick() - (Flags.backtrack_window / 1000)
local best, bestD = nil, math.huge for _, e in ipairs(buf) do local d =
math.abs(e.t - want) if d < bestD then bestD = d; best = e end end if
not best then return nowPos end return nowPos + (best.cf.Position -
nowPos) * 0.85 end
